async function migrarLocalStorageParaSupabase() { console.log("🚀 Iniciando migração do localStorage para o Supabase..."); let totalMigrados = 0; // Identifica quais manuais existem (freela, vagas, concursos) const manuais = ['freela', 'vagas', 'concursos']; for (let i = 0; i < localStorage.length; i++) { let chave = localStorage.key(i); let valorBruto = localStorage.getItem(chave); if (!valorBruto) continue; let conteudoObj; try { conteudoObj = JSON.parse(valorBruto); } catch (e) { // Se não for JSON válido, guarda como string dentro de um objeto conteudoObj = { texto_direto: valorBruto }; } // Tenta descobrir a qual manual essa chave pertence pelo prefixo ou nome let tipoManualDetectado = 'freela'; // padrão for (let m of manuais) { if (chave.includes(m)) { tipoManualDetectado = m; break; } } // Envia para o Supabase usando a mesma lógica de upsert const { error } = await dbSupabase .from('dados_cerebro') .upsert({ tipo_manual: tipoManualDetectado, chave_dado: chave, conteudo: conteudoObj, atualizado_em: new Date() }, { onConflict: ['tipo_manual', 'chave_dado'] }); if (error) { console.error(`❌ Erro ao migrar a chave "${chave}":`, error); } else { totalMigrados++; console.log(`✅ Migrado com sucesso: ${chave} (Manual: ${tipoManualDetectado})`); } } console.log(`✨ Migração concluída! Total de chaves migradas: ${totalMigrados}`); alert(`Migração finalizada! ${totalMigrados} registros foram enviados para o Supabase.`); } // Executa a função migrarLocalStorageParaSupabase();
